SELECT to_char(created_at AT TIME ZONE 'Asia/Jakarta', 'YYYY-MM-DD') AS day,
       count(*),
       sum(total)::bigint,
       round(avg(total))::bigint
FROM transactions
WHERE outlet_id = 1 AND status = 'PAID'
